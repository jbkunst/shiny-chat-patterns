App 00 · Conectar inputs y outputs

library(shiny)
library(bslib)
library(forecast)

# data --------------------------------------------------------------------
pasajeros <- utils::read.csv(file.path("..", "data", "air-passengers.csv"))

# user interface ----------------------------------------------------------
ui <- page_sidebar(
  title = "App 00 · La estructura mínima de una app",
  sidebar = sidebar(
    textInput("title", "Título", value = "Pasajeros aéreos"),
    sliderInput("n", "Cantidad de puntos", min = 24, max = 144, value = 48, step = 24),
    checkboxInput("forecast", "Mostrar pronóstico", value = FALSE)
  ),
  card(
    card_header(textOutput("plot_title")),
    plotOutput("plot")
  )
)

# server ------------------------------------------------------------------
server <- function(input, output, session) {
  
  output$plot_title <- renderText(paste0(input$title, " (", input$n, " observaciones)"))

  output$plot <- renderPlot({
    series <- ts(head(pasajeros$pasajeros, input$n), start = c(1949, 1), frequency = 12)

    if (input$forecast) {
      plot(
        forecast::forecast(series),
        col = "#0E4F5A", fcol = "#0E4F5A", shadecols = c("#DCECEE", "#A9CED2"),
        main = "", xlab = NULL, ylab = "Pasajeros (miles)"
      )
    } else {
      plot(
        series,
        col = "#0E4F5A", lwd = 2, main = "", xlab = NULL, ylab = "Pasajeros (miles)"
      )
    }
  })
}

{ellmer}

{ellmer} conecta R con modelos de lenguaje para construir conversaciones con contexto.

Configuración de {ellmer}

Para usar OpenAI desde {ellmer}, configura la clave en un archivo .Renviron.

OPENAI_API_KEY=clave_de_openai

Después de crear o modificar el archivo, reinicia la sesión de R.

Reemplaza clave_de_openai por tu clave personal en el archivo .Renviron de tu proyecto.

Una conversación con {ellmer}

Antes de conectarlo a una aplicación, probamos el cliente y la conversación de forma aislada.

library(ellmer)

chat <- ellmer::chat_openai(model = "gpt-5-nano", system_prompt = "Eres un bot muy alegre.")

chat$chat("Salúdame")

live_console(chat)

El cliente conserva el historial entre llamadas a chat$chat(). La clave se lee desde el entorno.

{shinychat}

{shinychat} conecta la conversación de {ellmer} con la interfaz reactiva de {shiny}.

App 02 · Conversar sin acceso a los datos

El chat en la UI

En la UI, chat_ui() crea la conversación. El id chat la conecta con el server; messages muestra un saludo inicial y placeholder orienta la pregunta. El modelo todavía no recibe los datos del dashboard.

shinychat::chat_ui(
  "chat",
  messages = saludo,
  placeholder = "Pregunta algo sobre los países..."
)

El chat en el server

En el server, chat_openai() crea el cliente. observeEvent() escucha input$chat_user_input y chat_append() devuelve el stream al componente con el mismo id.

chat <- ellmer::chat_openai(...)

observeEvent(input$chat_user_input, {
  stream <- chat$stream_async(input$chat_user_input)
  shinychat::chat_append("chat", stream)
})

Una tool con {ellmer}

Consulta un pronóstico por ciudad y fecha. Copia y ejecuta el bloque completo en R.

library(ellmer)
library(httr2)

# tools -------------------------------------------------------------------
consultar_pronostico <- function(ciudad, fecha) {
  lugares <- request("https://geocoding-api.open-meteo.com/v1/search") |>
    req_url_query(name = trimws(ciudad), count = 1, language = "es") |>
    req_perform() |>
    resp_body_json()
  lugar <- lugares$results[[1]]

  pronostico <- request("https://api.open-meteo.com/v1/forecast") |>
    req_url_query(
      latitude = lugar$latitude, longitude = lugar$longitude,
      daily = "temperature_2m_min,temperature_2m_max,precipitation_probability_max",
      timezone = "UTC", start_date = fecha, end_date = fecha
    ) |>
    req_perform() |>
    resp_body_json(simplifyVector = TRUE)

  list(
    ciudad = lugar$name, region = lugar$admin1, pais = lugar$country, fecha = fecha,
    pronostico = pronostico$daily, unidades = pronostico$daily_units
  )
}

# prompt ------------------------------------------------------------------
fecha_ejemplo <- format(as.Date(format(Sys.time(), tz = "UTC")) + 1, "%Y-%m-%d")
pregunta <- sprintf("¿Lloverá en Madrid el %s? Indica también las temperaturas.", fecha_ejemplo)

# chat --------------------------------------------------------------------
# Prueba primero la función sin el modelo:
consultar_pronostico("Madrid", fecha_ejemplo)

chat <- ellmer::chat_openai(
  model = "gpt-5-nano",
  system_prompt = paste(
    "Responde en español. Usa consultar_pronostico; no inventes datos meteorológicos.",
    "Si no tienes una tool disponible para consultar el pronóstico, explica que no puedes obtenerlo."
  )
)

# Sin registrar la tool: el modelo no puede consultar la API.
chat$chat(pregunta)

# Ahora damos acceso a la función y describimos sus argumentos.
chat$register_tool(tool(
  consultar_pronostico,
  "Consulta el pronóstico de un día en una ciudad. Usa la primera coincidencia.",
  arguments = list(
    ciudad = type_string("Nombre de una ciudad, por ejemplo Madrid o Tokio."),
    fecha = type_string("Fecha AAAA-MM-DD, entre hoy y hoy + 15 días, UTC.")
  )
))

# Con la tool disponible: vuelve a pedir el pronóstico.
chat$chat(paste("Ahora consulta el pronóstico con la herramienta disponible.", pregunta))

validar_select <- function(consulta) {
  consulta <- trimws(consulta)
  if (!grepl("^SELECT\\b", consulta, ignore.case = TRUE)) stop("Solo se permiten consultas SELECT.")

  if (grepl(";", consulta, fixed = TRUE)) stop("Envía una sola consulta y no incluyas punto y coma.")

Dos tools con {ellmer}

«¿Lloverá en Madrid dentro de tres días?» → consultar la fecha actual y luego el pronóstico.

library(ellmer)
library(httr2)

# tools -------------------------------------------------------------------
consultar_pronostico <- function(ciudad, fecha) {
  if (length(ciudad) != 1L || is.na(ciudad) || !nzchar(trimws(ciudad))) {
    stop("Indica una ciudad.")
  }
  if (length(fecha) != 1L || is.na(fecha) || !grepl("^\\d{4}-\\d{2}-\\d{2}$", fecha)) {
    stop("Usa una fecha con formato AAAA-MM-DD.")
  }
  dia <- tryCatch(as.Date(fecha), error = function(e) NA)
  hoy <- as.Date(format(Sys.time(), tz = "UTC", format = "%Y-%m-%d"))
  if (is.na(dia) || dia < hoy || dia > hoy + 15) {
    stop("La fecha debe estar entre hoy y los próximos 15 días, UTC.")
  }

  lugares <- request("https://geocoding-api.open-meteo.com/v1/search") |>
    req_url_query(name = trimws(ciudad), count = 1, language = "es") |>
    req_timeout(20) |>
    req_perform() |>
    resp_body_json()
  if (is.null(lugares$results)) stop("No encontré esa ciudad.")
  lugar <- lugares$results[[1]]

  pronostico <- request("https://api.open-meteo.com/v1/forecast") |>
    req_url_query(
      latitude = lugar$latitude, longitude = lugar$longitude,
      daily = "temperature_2m_min,temperature_2m_max,precipitation_probability_max",
      timezone = "UTC", start_date = fecha, end_date = fecha
    ) |>
    req_timeout(20) |>
    req_perform() |>
    resp_body_json(simplifyVector = TRUE)

  list(
    ciudad = lugar$name, region = lugar$admin1, pais = lugar$country, fecha = fecha,
    pronostico = pronostico$daily, unidades = pronostico$daily_units
  )
}

consultar_fecha_actual <- function() {
  list(
    fecha = format(Sys.time(), tz = "UTC", format = "%Y-%m-%d"),
    zona_horaria = "UTC"
  )
}

mostrar_puntos <- function() {
  puntos <- data.frame(x = rnorm(12), y = rnorm(12))
  puntos[12, ] <- runif(2, min = 4, max = 6)
  puntos <- puntos[sample(nrow(puntos)), ]
  puntos$nombre <- LETTERS[seq_len(nrow(puntos))]

  dibujar <- function() {
    plot(
      puntos$x, puntos$y, pch = 19, col = "#0E4F5A",
      xlim = range(puntos$x) + c(-0.5, 0.5), ylim = range(puntos$y) + c(-0.5, 0.5),
      xlab = "X", ylab = "Y", main = "¿Qué punto se separa del grupo?"
    )
    text(puntos$x, puntos$y, labels = puntos$nombre, pos = 3)
  }

  if (interactive()) dibujar()

  archivo <- tempfile(fileext = ".png")
  png(archivo, width = 700, height = 500)
  tryCatch(dibujar(), finally = dev.off())
  on.exit(unlink(archivo))
  content_image_file(archivo, resize = "none")

App 08 · Mover la cámara mediante un proxy

ir_a_pais() mueve la cámara sin cambiar el filtro de países ni el indicador del mapa.

El proxy conserva el mapa al viajar. Los cambios de filtro o color usan el render reactivo.

ir_a_pais <- function(pais, zoom = 4) {
  ubicacion <- DBI::dbGetQuery(
    con,
    paste(
      "SELECT DISTINCT p.pais, g.longitud, g.latitud",
      "FROM paises p JOIN paises_geo g USING (codigo)",
      "WHERE lower(p.pais) = lower(?)"
    ),
    params = list(pais)
  )
  if (!nrow(ubicacion)) stop("No encontré ese país.")

  mapgl::maplibre_proxy("map", session) |>
    mapgl::fly_to(
      center = c(ubicacion$longitud[1], ubicacion$latitud[1]),
      zoom = zoom
    )

  paste("Mapa centrado en", ubicacion$pais[1])
}

    )
  ))

  chat$register_tool(tool(
    ir_a_pais,
    "Vuela hacia un país en el mapa.",
    arguments = list(
      pais = type_string("Nombre del país en español."),
      zoom = type_number("Nivel de zoom, normalmente entre 2 y 6.", required = FALSE)
    )
  ))

  observeEvent(input$chat_user_input, {
    stream <- chat$stream_async(input$chat_user_input, tool_mode = "sequential", stream = "content")
    shinychat::chat_append("chat", stream)
  })
}
