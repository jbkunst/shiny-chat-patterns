server <- function(input, output, session) {
  
  output$plot_title <- renderText(paste0(input$title, " (", input$n, " observaciones)"))

    series <- ts(head(pasajeros$pasajeros, input$n), start = c(1949, 1), frequency = 12)

        series,
        col = "#0E4F5A", lwd = 2, main = "", xlab = NULL, ylab = "Pasajeros (miles)"
      )
    }
  })
}

    tinyplot_add(data = data(), col = "#0E4F5A", cex = 1.5)
  })

  output$table <- renderTable(data(), striped = TRUE, hover = TRUE)
}

Una tool con {ellmer}

Script R/02_ellmer_tool.R: definimos una función, la usamos directamente y la registramos como tool.

library(ellmer)
library(httr2)

# tools -------------------------------------------------------------------
consultar_pronostico <- function(ciudad, fecha) {
  lugares <- request("https://geocoding-api.open-meteo.com/v1/search") |>
    req_url_query(name = trimws(ciudad), count = 1, language = "es") |>
    req_perform() |>
    resp_body_json()
  lugar <- lugares$results[[1]]

  pronostico <- request("https://api.open-meteo.com/v1/forecast") |>
    req_url_query(
      latitude = lugar$latitude, longitude = lugar$longitude,
      daily = "temperature_2m_min,temperature_2m_max,precipitation_probability_max",
      timezone = "UTC", start_date = fecha, end_date = fecha
    ) |>
    req_perform() |>
    resp_body_json(simplifyVector = TRUE)

  list(
    ciudad = lugar$name, region = lugar$admin1, pais = lugar$country, fecha = fecha,
    pronostico = pronostico$daily, unidades = pronostico$daily_units
  )
}

# prompt ------------------------------------------------------------------
fecha_ejemplo <- format(as.Date(format(Sys.time(), tz = "UTC")) + 1, "%Y-%m-%d")
pregunta <- sprintf("¿Lloverá en Madrid el %s? Indica también las temperaturas.", fecha_ejemplo)

# chat --------------------------------------------------------------------
# Prueba primero la función sin el modelo:
consultar_pronostico("Madrid", fecha_ejemplo)

chat <- ellmer::chat(
  "openai/gpt-5-nano",
  system_prompt = paste(
    "Responde en español. Usa consultar_pronostico; no inventes datos meteorológicos.",
    "Si no tienes una tool disponible para consultar el pronóstico, explica que no puedes obtenerlo."
  )
)

# Sin registrar la tool: el modelo no puede consultar la API.
chat$chat(pregunta)

# Ahora damos acceso a la función y describimos sus argumentos.
chat$register_tool(tool(
  consultar_pronostico,
  "Consulta el pronóstico de un día en una ciudad. Usa la primera coincidencia.",
  arguments = list(
    ciudad = type_string("Nombre de una ciudad, por ejemplo Madrid o Tokio."),
    fecha = type_string("Fecha AAAA-MM-DD, entre hoy y hoy + 15 días, UTC.")
  )
))

# Con la tool disponible: vuelve a pedir el pronóstico.
chat$chat(paste("Ahora consulta el pronóstico con la herramienta disponible.", pregunta))

consultar_pronostico <- function(ciudad, fecha) {
  if (length(ciudad) != 1L || is.na(ciudad) || !nzchar(trimws(ciudad))) {
    stop("Indica una ciudad.")
  }
  if (length(fecha) != 1L || is.na(fecha) || !grepl("^\\d{4}-\\d{2}-\\d{2}$", fecha)) {
    stop("Usa una fecha con formato AAAA-MM-DD.")
  }
  dia <- tryCatch(as.Date(fecha), error = function(e) NA)
  hoy <- as.Date(format(Sys.time(), tz = "UTC", format = "%Y-%m-%d"))
  if (is.na(dia) || dia < hoy || dia > hoy + 15) {
    stop("La fecha debe estar entre hoy y los próximos 15 días, UTC.")
  }

  lugares <- request("https://geocoding-api.open-meteo.com/v1/search") |>
    req_url_query(name = trimws(ciudad), count = 1, language = "es") |>
    req_timeout(20) |>
    req_perform() |>
    resp_body_json()
  if (is.null(lugares$results)) stop("No encontré esa ciudad.")
  lugar <- lugares$results[[1]]

  pronostico <- request("https://api.open-meteo.com/v1/forecast") |>
    req_url_query(
      latitude = lugar$latitude, longitude = lugar$longitude,
      daily = "temperature_2m_min,temperature_2m_max,precipitation_probability_max",
      timezone = "UTC", start_date = fecha, end_date = fecha
    ) |>
    req_timeout(20) |>
    req_perform() |>
    resp_body_json(simplifyVector = TRUE)

  list(
    ciudad = lugar$name, region = lugar$admin1, pais = lugar$country, fecha = fecha,
    pronostico = pronostico$daily, unidades = pronostico$daily_units
  )
}

¿Qué agrega MCP?

MCP es un protocolo para descubrir y ejecutar herramientas de otros servicios desde nuestra aplicación.

  • Hasta ahora definimos las tools en R; con MCP recibimos sus nombres, argumentos y descripciones del servidor.
  • {mcptools} las convierte en tools de {ellmer}. El modelo decide cuándo llamarlas.
  • El servidor consulta los datos; el modelo usa el resultado para responder.

Usaremos el servidor comunitario de cyanheads. La fuente es el FMI; el MCP no es un servicio oficial. No requiere un token propio. El modelo sigue necesitando OPENAI_API_KEY.

# server ------------------------------------------------------------------
server <- function(input, output, session) {
  filas_vacias <- tibble(pais = character(), anio = integer(), x = double(), y = double())
  data <- reactiveVal(list(datos = filas_vacias, variables = tibble(indicador = character())))

  output$n_paises <- renderText(n_distinct(data()$datos$pais))
  output$n_variables <- renderText(n_distinct(data()$variables$indicador))
  output$periodo <- renderText({
    df <- data()$datos
    if (nrow(df)) paste(unique(range(df$anio)), collapse = "–") else "—"
  })
  output$table <- renderTable(data()$datos, digits = 4, striped = TRUE, hover = TRUE)

  output$plot <- renderPlot({
    df <- data()$datos |> filter(!is.na(x), !is.na(y))
    if (!nrow(df)) return(ggplot() + xlim(0, 1) + ylim(0, 1) + labs(x = "X", y = "Y"))
    etiquetas <- with(data()$variables, setNames(str_c(nombre, " · ", unidad), columna))
    ggplot(df, aes(x, y)) + geom_point(color = "#0E4F5A", size = 3) +
      labs(x = etiquetas["x"], y = etiquetas["y"])
  })

  output$map <- mapgl::renderMaplibre({
    map <- mapgl::maplibre(style = mapgl::carto_style("positron")) |> mapgl::add_globe_control()
    mapa <- mundo |> inner_join(data()$datos, by = c("adm0_a3" = "pais")) |> filter(!is.na(y))
    if (!nrow(mapa)) return(map)
    info <- data()$variables |> filter(columna == "y")
    mapa <- mapa |> mutate(detalle = str_glue(
      "{name_es}<br>{info$nombre}: {round(y, 4)} {info$unidad}<br>Año: {anio}<br>FMI · WEO"
    ))
    rango <- range(mapa$y)
    if (diff(rango) == 0) rango <- rango + c(-0.5, 0.5)
    colores <- c("#dcebea", "#0E4F5A")
    map |>
      mapgl::add_fill_layer(
        id = "paises", source = mapa, fill_opacity = 0.8, fill_outline_color = "white", tooltip = "detalle",
        fill_color = mapgl::interpolate("y", values = rango, stops = colores)
      ) |>
      mapgl::add_legend(str_c(info$nombre, " · ", info$unidad), values = rango, colors = colores)
  })

  # tools -----------------------------------------------------------------
  guardar_datos <- function(datos, variables) {
    data(list(datos = bind_rows(filas_vacias, datos), variables = bind_rows(variables)))
    paste("Vista actualizada:", length(datos), "países.")
  }

  ver_datos <- function() isolate(data())

  # chat ------------------------------------------------------------------
  chat <- ellmer::chat("openai/gpt-5.6-luna", system_prompt = prompt_sistema)
  chat$set_tools(mcp_tools("mcp.json"))
  chat$register_tool(tool(
    guardar_datos, "Reemplaza la carga: mapa usa y; gráfico y tabla usan x e y.",
    arguments = list(
      datos = type_array(type_object(
        pais = type_string("Código ISO3 del país, sin agregados."),
        anio = type_integer("Año común de los valores de esta fila."),
        x = type_number("Eje X; null para un mapa o si falta el valor.", required = FALSE),
        y = type_number("Eje Y y color del mapa; null si falta el valor.", required = FALSE)
      ), "Una fila por país; valores reales del MCP a cuatro decimales."),
      variables = type_array(type_object(
        columna = type_enum(c("x", "y"), "Columna descrita; y también se usa en el mapa."),
        indicador = type_string("ID del indicador del MCP."),
        nombre = type_string("Nombre del indicador."),
        unidad = type_string("Unidad original de la fuente.")
      ), "Metadatos de cada columna utilizada.")
    )
  ))
  chat$register_tool(tool(ver_datos, "Lee la carga actual, sus nombres y unidades, sin modificarla."))
  observeEvent(input$chat_user_input, {
    stream <- chat$stream_async(!!!input$chat_user_input, tool_mode = "sequential", stream = "content")
    shinychat::chat_append("chat", stream)
  })
}

Q&A · ¿Qué modelos admiten tools?

Estos modelos sí pueden solicitar llamadas a funciones mediante su API. Ejemplos a noviembre de 2026:

  • OpenAI: GPT-5 nano y GPT-5.6 Luna.
  • Anthropic: Claude Haiku 5.5, Sonnet 5.5 y Opus 5.5.
  • Google: Gemini 3.8 Flash y Gemini 3.1 Pro (preview).

La app debe registrar las tools. Que el modelo las soporte no garantiza que siempre las use correctamente.

Q&A para responder preguntas del público, con respuestas preparadas; no es una actividad interactiva. La lista da ejemplos confirmados en documentación oficial, no una lista exhaustiva ni un ranking de calidad. El soporte exige un modelo, una API y un cliente compatibles; el acceso también depende de la cuenta. Tanto GPT-5 nano como GPT-5.6 Luna admiten function calling. La diferencia observada en el taller es de comportamiento, no que nano carezca de soporte ni que Luna conozca nuestras funciones sin registrarlas. Ficha adicional de GPT-5 nano: https://developers.openai.com/api/docs/models/gpt-5-nano. La capacidad de llamar funciones no implica admitir imágenes como resultados de esas funciones; esa compatibilidad debe revisarse por separado para ejemplos como R/04_ellmer_imagen.R.
