App 00 · Conectar inputs y outputs

library(shiny)
library(bslib)
library(forecast)

# data --------------------------------------------------------------------
pasajeros <- utils::read.csv(file.path("..", "data", "air-passengers.csv"))

# user interface ----------------------------------------------------------
ui <- page_sidebar(
  title = "App 00 · La estructura mínima de una app",
  sidebar = sidebar(
    textInput("title", "Título", value = "Pasajeros aéreos"),
    sliderInput("n", "Cantidad de puntos", min = 24, max = 144, value = 48, step = 24),
    checkboxInput("forecast", "Mostrar pronóstico", value = FALSE)
  ),
  card(
    card_header(textOutput("plot_title")),
    plotOutput("plot")
  )
)

# server ------------------------------------------------------------------
server <- function(input, output, session) {
  
  output$plot_title <- renderText(paste0(input$title, " (", input$n, " observaciones)"))

  output$plot <- renderPlot({
    series <- ts(head(pasajeros$pasajeros, input$n), start = c(1949, 1), frequency = 12)

    if (input$forecast) {
      plot(
        forecast::forecast(series),
        col = "#0E4F5A", fcol = "#0E4F5A", shadecols = c("#DCECEE", "#A9CED2"),
        main = "", xlab = NULL, ylab = "Pasajeros (miles)"
      )
    } else {
      plot(
        series,
        col = "#0E4F5A", lwd = 2, main = "", xlab = NULL, ylab = "Pasajeros (miles)"
      )
    }
  })
}

library(shiny)
library(bslib)
library(tinyplot)

# data --------------------------------------------------------------------
paises <- utils::read.csv(file.path("..", "data", "paises.csv"), fileEncoding = "UTF-8")
continentes <- c("Todos", sort(unique(paises$continente)))

# user interface ----------------------------------------------------------
ui <- page_sidebar(
  title = "App 01 · Dashboard reactivo",
  sidebar = sidebar(selectInput("continente", "Continente", choices = continentes), width = 400),
  layout_columns(
    value_box("Países", textOutput("n_filas"), showcase = icon("earth-americas"), theme = "text-primary"),
    value_box("Esperanza de vida", textOutput("vida"), showcase = icon("heart-pulse"), theme="text-primary"),
    value_box("Población", textOutput("poblacion"), showcase = icon("people-group"), theme = "text-primary"),
    card(plotOutput("plot")),
    card(tableOutput("table")),
    col_widths = c(4, 4, 4, 6, 6),
    row_heights = c(1, 3)
  )
)

# server ------------------------------------------------------------------
server <- function(input, output, session) {
  data <- reactive({
    if (input$continente == "Todos") paises
    else paises[paises$continente == input$continente, ]
  })

  output$n_filas   <- renderText(nrow(data()))
  output$vida      <- renderText(paste0(round(mean(data()$esperanza_de_vida), 1), " años"))
  output$poblacion <- renderText(paste0(round(sum(data()$poblacion) / 1e6), " millones"))

  output$plot <- renderPlot({
    tinyplot(esperanza_de_vida ~ pib_per_capita, data = paises, pch = 19, col = "#d2d2d2", log = "x")
    tinyplot_add(data = data(), col = "#0E4F5A", cex = 1.5)
  })

  output$table <- renderTable(data(), striped = TRUE, hover = TRUE)
}

{ellmer}

{ellmer} conecta R con modelos de lenguaje para construir conversaciones con contexto.

Configuración de {ellmer}

Para usar OpenAI desde {ellmer}, configura la clave en un archivo .Renviron.

OPENAI_API_KEY=clave_de_openai

Después de crear o modificar el archivo, reinicia la sesión de R.

No incluyas una clave real en la presentación ni subas .Renviron al repositorio.

Una conversación con {ellmer}

Antes de conectarlo a una aplicación, probamos el cliente y la conversación de forma aislada.

library(ellmer)

chat <- ellmer::chat_openai(model = "gpt-5-nano", system_prompt = "Eres un bot muy alegre.")

chat$chat("Salúdame")

chat$chat("¿Cuánto es 2 elevado a 4?")

live_console(chat)

El cliente conserva el historial entre llamadas a chat$chat(). La clave se lee desde el entorno.

{shinychat}

{shinychat} conecta la conversación de {ellmer} con la interfaz reactiva de {shiny}.

App 02 · Conversar sin acceso a los datos

El chat en la UI

En la UI, chat_ui() crea la conversación. El id chat la conecta con el server; messages muestra un saludo inicial y placeholder orienta la pregunta. El modelo todavía no recibe los datos del dashboard.

shinychat::chat_ui(
  "chat",
  messages = saludo,
  placeholder = "Pregunta algo sobre los países..."
)

El chat en el server

En el server, chat_openai() crea el cliente. observeEvent() escucha input$chat_user_input y chat_append() devuelve el stream al componente con el mismo id.

chat <- ellmer::chat_openai(...)

observeEvent(input$chat_user_input, {
  stream <- chat$stream_async(input$chat_user_input)
  shinychat::chat_append("chat", stream)
})

App 03 · Leer un resumen de los datos visibles

La tool calcula un resumen con campos predefinidos a partir de los países visibles.

isolate(data()) lee el estado al ejecutar la tool. Esta consulta no modifica el dashboard.

resumir_datos_visibles <- function() {
  df <- isolate(data())
  list(
    continente = isolate(input$continente),
    registros = nrow(df),
    poblacion_total = sum(df$poblacion),
    esperanza_promedio = round(mean(df$esperanza_de_vida), 1),
    pib_promedio = round(mean(df$pib_per_capita))
  )
}

chat$register_tool(tool(
  resumir_datos_visibles,
  "Resume los países visibles en el dashboard."

App 04 · Pasar del resumen a consultas SQL

La tool consulta toda la tabla paises; el filtro manual del dashboard no limita esa consulta.

El código exige SELECT, la palabra paises y ningún punto y coma: es una validación básica.

El prompt pide hasta 20 filas al listar registros; ese límite no está impuesto por el código.

consultar_paises <- function(consulta) {
  DBI::dbGetQuery(con, validar_select(consulta))
}

chat$register_tool(tool(
  consultar_paises,
  "Ejecuta una consulta SELECT sobre la tabla paises.",
  arguments = list(
    consulta = type_string("Consulta SELECT sin punto y coma.")
  )
))

validar_select <- function(consulta) {
  consulta <- trimws(consulta)
  if (!grepl("^SELECT\\b", consulta, ignore.case = TRUE)) stop("Solo se permiten consultas SELECT.")

App 05 · Pasar de consultar a actualizar la vista

Consultar devuelve datos al chat. Actualizar cambia la consulta reactiva que alimenta el dashboard.

Antes de cambiar el estado, la tool comprueba que el resultado conserve todas las columnas.

actualizar_dashboard <- function(consulta) {
  consulta <- validar_select(consulta)
  resultado <- DBI::dbGetQuery(con, consulta)
  if (!all(names(paises) %in% names(resultado))) stop("La consulta debe devolver todas las columnas.")

  consulta_actual(consulta)

  list(mensaje = "Dashboard actualizado.", registros = nrow(resultado))
}

App 05 · Cómo se propaga el cambio

Al ejecutar consulta_actual(consulta), Shiny invalida los cálculos que dependen de ese estado.

consulta_actual <- reactiveVal("SELECT * FROM paises")
data <- reactive(DBI::dbGetQuery(con, consulta_actual()))

Los outputs dependen de data() y vuelven a calcularse para mostrar el resultado de la nueva consulta.

server <- function(input, output, session) {
  consulta_actual <- reactiveVal("SELECT * FROM paises")

  data <- reactive(DBI::dbGetQuery(con, consulta_actual()))

  output$n_filas   <- renderText(nrow(data()))
  output$vida      <- renderText(paste0(round(mean(data()$esperanza_de_vida), 1), " años"))
  output$poblacion <- renderText(paste0(round(sum(data()$poblacion) / 1e6), " millones"))

  output$plot <- renderPlot({
    tinyplot(esperanza_de_vida ~ pib_per_capita, data = paises, pch = 19, col = "#d2d2d2", log = "x")
    tinyplot_add(data = data(), col = "#0E4F5A", cex = 1.5)
  })

  output$table <- renderTable(data(), striped = TRUE, hover = TRUE)

  # tools -----------------------------------------------------------------
  consultar_paises <- function(consulta) {
    DBI::dbGetQuery(con, validar_select(consulta))
  }

  actualizar_dashboard <- function(consulta) {
    consulta <- validar_select(consulta)
    resultado <- DBI::dbGetQuery(con, consulta)
    if (!all(names(paises) %in% names(resultado))) stop("La consulta debe devolver todas las columnas.")

    consulta_actual(consulta)
    list(mensaje = "Dashboard actualizado.", registros = nrow(resultado))
  }

  mostrar_tabla <- function(n, titulo) {
    n <- max(1L, min(as.integer(n), 100L))
    df <- isolate(data())
    resultado <- head(df, n)
    output$tabla_modal <- renderTable(resultado, striped = TRUE, hover = TRUE)

    showModal(modalDialog(
      title = titulo, tableOutput("tabla_modal"), size = "l", easyClose = TRUE,
      footer = modalButton("Cerrar")
    ))

    resultado
  }

  mostrar_grafico <- function(x, y, color, titulo) {
    df <- isolate(data())
    columnas <- c(x, y, color)
    columnas <- columnas[nzchar(columnas)]
    if (!all(columnas %in% names(df))) stop("Una o más columnas no existen.")
    if (!nrow(df)) stop("No hay datos para graficar.")
    formula <- as.formula(paste(y, "~", x, if (nzchar(color)) paste("|", color) else ""))

    dibujar <- function() {
      if (nzchar(color)) tinyplot(formula, data = df, pch = 19, palette = "Blues 3")
      else tinyplot(formula, data = df, pch = 19, col = "#0E4F5A")
    }

    output$grafico_modal <- renderPlot(dibujar())

    showModal(modalDialog(
      title = titulo, plotOutput("grafico_modal", height = "500px"), size = "l", easyClose = TRUE,
      footer = modalButton("Cerrar")
    ))

    archivo <- tempfile(fileext = ".png")
    grDevices::png(archivo, width = 768, height = 500)
    tryCatch(dibujar(), finally = grDevices::dev.off())
    on.exit(unlink(archivo))
    content_image_file(archivo, resize = "none")
  }

  # chat ------------------------------------------------------------------
  chat <- ellmer::chat_openai(model = "gpt-5-nano", system_prompt = prompt_sistema)

  chat$register_tool(tool(
    consultar_paises,
    "Ejecuta una consulta SELECT de solo lectura sobre la tabla DuckDB paises.",
    arguments = list(
      consulta = type_string("Consulta DuckDB SELECT sobre la tabla paises, sin punto y coma.")
    )
  ))

  chat$register_tool(tool(
    actualizar_dashboard,
    "Filtra, ordena o reinicia los datos reactivos mostrados en el dashboard.",
    arguments = list(
      consulta = type_string("SELECT * FROM paises con WHERE u ORDER BY, sin punto y coma.")
    )
  ))

  chat$register_tool(tool(
    mostrar_tabla,
    "Abre en un modal una tabla con los datos visibles en el dashboard.",
    arguments = list(
      n = type_integer("Cantidad de filas, entre 1 y 100."),
      titulo = type_string("Título breve del modal.")
    )
  ))

  chat$register_tool(tool(
    mostrar_grafico,
    "Grafica los datos visibles en un modal y devuelve la imagen al modelo.",
    arguments = list(
      x = type_enum(c("poblacion", "esperanza_de_vida", "pib_per_capita"), "Columna para el eje x."),
      y = type_enum(c("poblacion", "esperanza_de_vida", "pib_per_capita"), "Columna para el eje y."),
      color = type_enum(c("", "continente"), "Columna para el color, o ninguna."),
      titulo = type_string("Título breve del modal.")
    )
  ))

  observeEvent(input$chat_user_input, {
    stream <- chat$stream_async(input$chat_user_input, tool_mode = "sequential", stream = "content")
    shinychat::chat_append("chat", stream)
  })
}

App 08 · Mover la cámara mediante un proxy

ir_a_pais() mueve la cámara sin cambiar el filtro de países ni el indicador del mapa.

El proxy conserva el mapa al viajar. Los cambios de filtro o color usan el render reactivo.

ir_a_pais <- function(pais, zoom = 4) {
  ubicacion <- DBI::dbGetQuery(
    con,
    paste(
      "SELECT DISTINCT p.pais, g.longitud, g.latitud",
      "FROM paises p JOIN paises_geo g USING (codigo)",
      "WHERE lower(p.pais) = lower(?)"
    ),
    params = list(pais)
  )
  if (!nrow(ubicacion)) stop("No encontré ese país.")

  mapgl::maplibre_proxy("map", session) |>
    mapgl::fly_to(
      center = c(ubicacion$longitud[1], ubicacion$latitud[1]),
      zoom = zoom
    )

  paste("Mapa centrado en", ubicacion$pais[1])
}

server <- function(input, output, session) {
  consulta_actual <- reactiveVal("SELECT * FROM paises")
  variable_mapa <- reactiveVal("esperanza_de_vida")

  data <- reactive({
    consulta <- paste(
      "SELECT q.*, ST_AsText(g.geometry) AS wkt",
      paste0("FROM (", consulta_actual(), ") q"),
      "JOIN paises_geo g USING (codigo)"
    )
    DBI::dbGetQuery(con, consulta) |> sf::st_as_sf(wkt = "wkt", crs = 4326)
  })

  output$n_filas   <- renderText(nrow(data()))
  output$vida      <- renderText(paste0(round(mean(data()$esperanza_de_vida), 1), " años"))
  output$poblacion <- renderText(paste0(round(sum(data()$poblacion) / 1e6), " millones"))

  output$plot <- renderPlot({
    tinyplot(esperanza_de_vida ~ pib_per_capita, data = paises, pch = 19, col = "#d2d2d2", log = "x")
    tinyplot_add(data = sf::st_drop_geometry(data()), col = "#0E4F5A", cex = 1.5)
  })

  output$map <- mapgl::renderMaplibre({
    df <- data()
    req(nrow(df))
    df$valor_mapa <- switch(variable_mapa(),
      poblacion = log10(df$poblacion),
      pib_per_capita = log10(df$pib_per_capita),
      df$esperanza_de_vida
    )
    valores <- rangos_mapa[[variable_mapa()]]
    colores <- mapgl::interpolate("valor_mapa", values = valores, stops = c("#dcebea", "#0E4F5A"))

    mapgl::maplibre(style = mapgl::carto_style("positron"), bounds = df) |>
      mapgl::add_fill_layer(
        id = "paises", source = df, fill_color = colores, fill_opacity = 0.8,
        fill_outline_color = "white", tooltip = "pais"
      ) |>
      mapgl::add_globe_control()
  })

  # tools -----------------------------------------------------------------
  consultar_paises <- function(consulta) {
    DBI::dbGetQuery(con, validar_select(consulta))
  }

    consulta <- validar_select(consulta)

      registros = nrow(resultado),
      esperanza_promedio = round(mean(resultado$esperanza_de_vida), 1),
      poblacion_total = sum(resultado$poblacion),
      pib_promedio = round(mean(resultado$pib_per_capita))
    )
  }

    variable_mapa(variable)
    paste("Mapa coloreado por", variable)
  }

    ubicacion <- DBI::dbGetQuery(con, paste(
      "SELECT DISTINCT p.pais, g.longitud, g.latitud",
      "FROM paises p JOIN paises_geo g USING (codigo)",
